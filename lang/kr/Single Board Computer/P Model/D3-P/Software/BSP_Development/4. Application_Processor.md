다음은 메인 코어용으로 Yocto에서 U-boot를 빌드할 때 적용되는 기능을 설명합니다.

```
$ cat linux-telechips.inc

SUMMARY = "Universal Boot Loader for embedded devices"
HOMEPAGE = "http://www.denx.de/wiki/U-Boot/WebHome"
SECTION = "bootloaders"
PROVIDES = "virtual/bootloader"

LICENSE = "GPLv2+"
LIC_FILES_CHKSUM = "file://Licenses/README;md5=30503fd321432fc713238f582193b78e"

SRC_URI = "${TELECHIPS_AUTOMOTIVE_BSP_GIT}/u-boot.git;protocol=${ALS_GIT_PROTOCOL};branch=${ALS_BRANCH}"

SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'hud-display', 'file://add-dp.cfg', '', d)}"
SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'early-camera', 'file://add-earlycam.cfg', '', d)}"
SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'gpu-vz', 'file://gpu-vz.cfg', '', d)}"

# DP to HDMI (1920x1080)
SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'dp2hdmi', 'file://add-dp2hdmi.cfg', '', d)}"

SRC_URI_append_tcc803x = " ${@bb.utils.contains('INVITE_PLATFORM', 'with-subcore', 'file://add-subcore.cfg', '', d)}"
SRC_URI_append_tcc803x = " ${@bb.utils.contains('INVITE_PLATFORM', 'cluster-display', 'file://add-dlvds.cfg', '', d)}"

#SRCREV = "3965996feb533416e55780924154f16fd5dc7735"
SRCREV = "${AUTOREV}"

S = "${WORKDIR}/git"
B = "${S}"

PACKAGE_ARCH = "${MACHINE_ARCH}"

inherit uboot-config deploy

DEPENDS += "bison-native"

EXTRA_OEMAKE = 'CROSS_COMPILE=${TARGET_PREFIX} CC="${TARGET_PREFIX}gcc ${TOOLCHAIN_OPTIONS}" V=1'
EXTRA_OEMAKE += 'HOSTCC="${BUILD_CC} ${BUILD_CFLAGS} ${BUILD_LDFLAGS}"'
EXTRA_OEMAKE += 'STAGING_INCDIR=${STAGING_INCDIR_NATIVE} STAGING_LIBDIR=${STAGING_LIBDIR_NATIVE}'

UBOOT_ARCH_ARGS_arm = "ARCH=arm"
UBOOT_ARCH_ARGS_aarch64 = "ARCH=arm64 "

SNOR_BOOT_NAME = "${TCC_ARCH_FAMILY}_snor_boot.rom"

PATCHTOOL = "git"

def find_cfgs(d):
    sources=src_patches(d, True)
    sources_list=[]
    for s in sources:
        if s.endswith('.cfg'):
            sources_list.append(s)

    return sources_list

do_configure_append() {
        if [ "${@bb.utils.contains('DISTRO_FEATURES', 'ld-is-gold', 'ld-is-gold', '', d)}" = "ld-is-gold" ] ; then
                sed -i 's/$(CROSS_COMPILE)ld$/$(CROSS_COMPILE)ld.bfd/g' ${S}/config.mk
        fi

        unset LDFLAGS
        unset CFLAGS
        unset CPPFLAGS
        if [ -n "${EXTERNALSRC}" ] ; then
                export KBUILD_OUTPUT=${EXTERNALSRC}/${MACHINE}/
        fi
        export ${UBOOT_ARCH_ARGS} DEVICE_TREE=${UBOOT_DEVICE_TREE}
        oe_runmake ${UBOOT_MACHINE}
        if [ -n "${EXTERNALSRC}" ] ; then
                ${S}/scripts/kconfig/merge_config.sh -O ${KBUILD_OUTPUT} -m ${KBUILD_OUTPUT}.config ${@" ".join(find_cfgs(d))}
        else
                ${S}/scripts/kconfig/merge_config.sh -m .config ${@" ".join(find_cfgs(d))}
        fi
}

do_compile() {
        unset LDFLAGS
        unset CFLAGS
        unset CPPFLAGS
        if [ -n "${EXTERNALSRC}" ] ; then
                export KBUILD_OUTPUT=${EXTERNALSRC}/${MACHINE}/
        fi
        export ${UBOOT_ARCH_ARGS} DEVICE_TREE=${UBOOT_DEVICE_TREE}

        oe_runmake
}

do_install() {
        install -d ${D}/boot

        if [ -n "${EXTERNALSRC}" ] ; then
                export KBUILD_OUTPUT=${EXTERNALSRC}/${MACHINE}/
                install -m 0644 ${KBUILD_OUTPUT}${UBOOT_NAME}.${UBOOT_SUFFIX}   ${D}/boot/${UBOOT_IMAGE}
        else
                install -m 0644 ${S}/${UBOOT_NAME}.${UBOOT_SUFFIX}      ${D}/boot/${UBOOT_IMAGE}
        fi

        if ${@bb.utils.contains('INVITE_PLATFORM', 'early-camera', 'true', 'false', d)}; then
                install -m 0644 ${S}/drivers/camera/splash/${SPLASH_IMAGE}  ${D}/boot/${SPLASH_IMAGE}
        fi
}

FILES_${PN} = "/boot"

do_deploy() {
    install -d ${DEPLOYDIR}
    install -m 0644 ${D}/boot/${UBOOT_IMAGE} ${DEPLOYDIR}

        cd ${DEPLOYDIR}
    rm -f ${UBOOT_BINARY} ${UBOOT_SYMLINK}
    ln -sf ${UBOOT_IMAGE} ${UBOOT_BINARY}
    ln -sf ${UBOOT_IMAGE} ${UBOOT_SYMLINK}
        cd -

        if ${@bb.utils.contains('INVITE_PLATFORM', 'early-camera', 'true', 'false', d)}; then
        install -m 0644 ${D}/boot/${SPLASH_IMAGE} ${DEPLOYDIR}
    fi
}

do_clean_extworkdir() {
        if [ -n "${EXTERNALSRC}" ]; then
                rm -rf ${EXTERNALSRC}/arch/*/include/asm/telechips
                rm -rf ${EXTERNALSRC}/${MACHINE}
        fi
}

addtask deploy before do_build after do_install
addtask clean_extworkdir after do_buildclean before do_clean
```

do_tc_make_image() {
        cd ${D}/${KERNEL_IMAGEDEST}

        touch ${RAMDISK_NAME}
        #compressed boot image
        ${STAGING_BINDIR_NATIVE}/tc-make-bootimg --kernel ${KERNEL_IMAGETYPE}-${KERNEL_VERSION} --ramdisk ${RAMDISK_NAME} --base ${KERNEL_BASE_ADDR} --kernel_offset ${KERNEL_OFFSET} --output ${BOOT_IMAGE} --cmdline ${CMDLINE}
        #uncompressed boot image
        ${STAGING_BINDIR_NATIVE}/tc-make-bootimg --kernel ${KERNEL_IMAGETYPE_UNCOMPRESSED}-${KERNEL_VERSION} --ramdisk ${RAMDISK_NAME} --base ${KERNEL_BASE_ADDR} --kernel_offset ${KERNEL_OFFSET} --output ${BOOT_IMAGE_UNCOMPRESSED} --cmdline ${CMDLINE}
        rm  ${RAMDISK_NAME}

        cd -
}

### 4.3.2.3 USB 2.0 DRD 드라이버

디바이스 기능과 호스트 기능을 모두 제공하는 USB 2.0 Dual-Role Device (DRD) 컨트롤러가 지원됩니다. USB 2.0 사양을 완전히 준수하는 호스트 전용 또는 디바이스 전용 컨트롤러로 구성할 수도 있습니다. USB 2.0 구성은 high-speed (HS, 480 Mbps), full-speed (FS, 12 Mbps) 및 low-speed (LS, 1.5 Mbps) 전송을 지원합니다. 또한 USB 1.1 full-speed/low-speed DRD로 구성할 수도 있습니다.

### 4.3.3 디스플레이

DP-to-HDMI 어댑터에는 패시브와 액티브의 두 가지 유형이 있습니다. TOPST는 액티브 어댑터만 지원합니다.

<p align="center">
    <img src="https://github.com/topst-development/Documentation/assets/161264431/e5287e06-624a-4824-a1b6-f3469887d195" >
</p>
<p align="center"><strong>그림 4.8 DP 커넥터 회로도 </strong>

TOPST는 기본적으로 DisplayPort를 사용합니다. 이 기능은 메인 코어에서 지원됩니다.

“Local.conf”의 **dp2hdmi** 기능을 사용하여 DisplayPort 또는 DP2HDMI를 선택하십시오.

'Local.conf' 설정에서 DisplayPort 설정이 지정되면 아래와 같이 U-Boot와 커널에 적용됩니다.

```
$ cat build/tcc8050-main/conf/local.conf

# DP to HDMI (1920x1080)
# if you are not defined, set the DP mode 
#INVITE_PLATFORM += "dp2hdmi"
```

<br/>

다음은 U-Boot에서 **dp2hdmi**를 사용하기 위한 설정입니다.

```
$ cat poky/meta-telechips-bsp/recipes-bsp/u-boot/u-boot-tcc.bb 

# DP to HDMI (1920x1080)
SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'dp2hdmi', 'file://add-dp2hdmi.cfg', '', d)}"
```

<br/>

다음은 커널에서 **DP** 또는 **dp2hdmi**를 사용하기 위한 설정입니다.

```
$ cat poky/meta-telechips-bsp/recipes-kernel/linux/linux-telechips.inc 

if ${@bb.utils.contains('INVITE_PLATFORM', 'dp2hdmi', 'true', 'false', d)}; then
		# DP to HDMI (1920x1080)
		echo "CONFIG_DRM_TCC_LCD_VIC=16"			>> ${WORKDIR}/defconfig
	else
		# DP mode
		echo "CONFIG_DRM_TCC_LCD_VIC=0"				>> ${WORKDIR}/defconfig
	fi
}
```

<br/>

다음은 커널에서 1920x1080 60Hz를 지원하는 모니터를 위한 설정입니다.

해상도가 다른 모니터의 경우, 해당 모니터에 맞게 다음을 변경해야 합니다.

display-timings {
			native-mode = <&timing_dpv14_0>;

				hactive = <1920>;
				vactive = <1080>;
				hfront-porch = <88>;
				hback-porch = <148>;
				hsync-len = <44>;
				hsync-active = <1>;
				vback-porch = <36>;
				vfront-porch = <4>;
				vsync-len = <5>;
				vsync-active = <1>;

I2C는 호스트 어댑터 (또는 호스트 컨트롤러)와 클라이언트 디바이스 (또는 슬레이브) 간에 통신이 이루어지는 마스터-슬레이브 프로토콜이며, 클럭과 양방향 데이터 전송을 위한 두 개의 라인을 가집니다. 해당 라인은 Serial Clock (SCL)과 Serial Data (SDA)라고 합니다. 유저 스페이스 접근을 위해 I2C 마스터 디바이스 노드 (/dev/i2c-x 등)가 생성됩니다.

### 4.3.8.2.2 GPIO sysfs 인터페이스

커널은 사용자 공간에서 GPIO를 사용할 수 있도록 **sysfs**를 통한 인터페이스를 제공합니다. GPIO **sysfs interface** 사용 방법에 대한 문서는 커널의 “Documentation/gpio/sysfs.txt” 파일에 있습니다.

**sysfs**를 포함하려면 **menuconfig**에서 다음을 추가하십시오.

```
$ bitbake virtual/kernel -c menuconfig

Device Drivers ---> 
GPIO Support ---> 
[*] /sys/class/gpio/… (sysfs interface)
```

- GPIO 번호

사용 중인 칩에 맞는 pinctrl 파일이 제공됩니다. 플랫폼이 TCC8050인 경우 “tcc8050-pinctrl.dtsi”를 사용해야 합니다. 이 파일은 **gpio a/b/c/d/e** 등을 사용하도록 GPIO를 정의합니다. **reg** 값이 **sysfs**에서의 값을 결정합니다.

예를 들어 **gpio a**의 값은 32이며 **reg** 값에 정의되어 있습니다. **gpio a**는 GPIO의 첫 번째이므로 **sysfs**에 0부터 31까지 입력할 수 있습니다.

**gpio b**를 전달하는 경우 **gpio a**가 0부터 31까지 사용하므로 **gpio b 0**은 32부터 시작합니다.

예를 들어 **gpio b 3**을 사용하는 경우 **sysfs**에는 32 + 3으로 입력해야 합니다.

각 GPIO에 할당된 GPIO 번호는 아래에 빨간색으로 표시되어 있습니다. GPIO는 GPIO 값을 0에서 31까지 증가시켜 **sysfs**에 입력합니다.

```
gpa: gpa { 
gpio-controller; 
#gpio-cells = <2>; 
reg = <0x0 0 32>; 
}; 

gpb: gpb { 
gpio-controller; 
#gpio-cells = <2>; 
reg = <0x40 0 32>; 
}; 

gpc: gpc { 
gpio-controller; 
#gpio-cells = <2>; 
reg = <0x80 0 32>; 
};

```

### 4.3.8.2.3 GPIO dev 인터페이스

- GPIO dev 인터페이스 준비 (**libgpiod**)

GPIO sysfs 인터페이스는 더 이상 사용되지 않으며 새로운 GPIO dev 인터페이스가 병합되었습니다. GPIO dev 인터페이스에는 GPIO chardev용 C 라이브러리 및 도구인 libgpiody가 필요하며, GPIO dev 인터페이스는 /dev 디렉터리에 있는 gpiochip0, gpiochip1, ghiochip2 등의 이름을 가진 디바이스 노드를 사용합니다.

다음 예제는 간단하며 GPIO 제어의 기본 기능만 다룹니다.

- Raspberry GPIO 핀 예제 (**RPI GPIO_C22 PIN7**)

```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <string.h>

// RPI GPIO_C22 PIN7

int main(int argc, char *argv[])
{
    int fd;
    char buf[64];
    int pin = 22; // choose a GPIO pin
    int value = 1; // choose a value (0 or 1)

    // open GPIO chardev number 0. GPIO group C. 
    snprintf(buf, sizeof(buf), "/dev/gpiochip2");
    fd = open(buf, O_RDWR);
    if (fd < 0) {
        perror("Failed to open gpiochip0");
        exit(1);
    }

// get the desired line of GPIO chip. GPIO_C22. 
    snprintf(buf, sizeof(buf), "gpio%d/direction", pin);
    if (write(fd, "out", 4) < 0) {
        perror("Failed to set direction of gpio pin");
        exit(1);
    }

    // set the value of the GPIO pin. GPIO_C22.
    snprintf(buf, sizeof(buf), "gpio%d/value", pin);
    if (write(fd, value ? "1" : "0", 2) < 0) {
        perror("Failed to set value of gpio pin");
        exit(1);
    }

    // close the GPIO chip device
    close(fd);

    return 0;
}

```

### 4.3.10 PWM

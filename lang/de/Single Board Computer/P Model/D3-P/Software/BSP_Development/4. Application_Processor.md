### 4.1.1.1.1 So finden Sie das U-Boot-Rezept und den Quellcode in Yocto

**u-boot-tcc.bb** ist ein Rezept, das durch Hinzufügen von Funktionalität für TOPST zum U-Boot-Open-Source-Code erstellt wurde.

Im Folgenden werden die Funktionen beschrieben, die beim Erstellen von U-boot in Yocto für den Hauptkern angewendet werden.

```
$ cat linux-telechips.inc

SUMMARY = "Universal Boot Loader for embedded devices"
HOMEPAGE = "http://www.denx.de/wiki/U-Boot/WebHome"
SECTION = "bootloaders"
PROVIDES = "virtual/bootloader"

LICENSE = "GPLv2+"
LIC_FILES_CHKSUM = "file://Licenses/README;md5=30503fd321432fc713238f582193b78e"

SRC_URI = "${TELECHIPS_AUTOMOTIVE_BSP_GIT}/u-boot.git;protocol=${ALS_GIT_PROTOCOL};branch=${ALS_BRANCH}"

SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'hud-display', 'file://add-dp.cfg', '', d)}"
SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'early-camera', 'file://add-earlycam.cfg', '', d)}"
SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'gpu-vz', 'file://gpu-vz.cfg', '', d)}"

# DP to HDMI (1920x1080)
SRC_URI += "${@bb.utils.contains('INVITE_PLATFORM', 'dp2hdmi', 'file://add-dp2hdmi.cfg', '', d)}"

SRC_URI_append_tcc803x = " ${@bb.utils.contains('INVITE_PLATFORM', 'with-subcore', 'file://add-subcore.cfg', '', d)}"
SRC_URI_append_tcc803x = " ${@bb.utils.contains('INVITE_PLATFORM', 'cluster-display', 'file://add-dlvds.cfg', '', d)}"

#SRCREV = "3965996feb533416e55780924154f16fd5dc7735"
SRCREV = "${AUTOREV}"

S = "${WORKDIR}/git"
B = "${S}"

PACKAGE_ARCH = "${MACHINE_ARCH}"

inherit uboot-config deploy

DEPENDS += "bison-native"

EXTRA_OEMAKE = 'CROSS_COMPILE=${TARGET_PREFIX} CC="${TARGET_PREFIX}gcc ${TOOLCHAIN_OPTIONS}" V=1'
EXTRA_OEMAKE += 'HOSTCC="${BUILD_CC} ${BUILD_CFLAGS} ${BUILD_LDFLAGS}"'
EXTRA_OEMAKE += 'STAGING_INCDIR=${STAGING_INCDIR_NATIVE} STAGING_LIBDIR=${STAGING_LIBDIR_NATIVE}'

UBOOT_ARCH_ARGS_arm = "ARCH=arm"
UBOOT_ARCH_ARGS_aarch64 = "ARCH=arm64 "

SNOR_BOOT_NAME = "${TCC_ARCH_FAMILY}_snor_boot.rom"

PATCHTOOL = "git"

def find_cfgs(d):
    sources=src_patches(d, True)
    sources_list=[]
    for s in sources:
        if s.endswith('.cfg'):
            sources_list.append(s)

    return sources_list

do_configure_append() {
        if [ "${@bb.utils.contains('DISTRO_FEATURES', 'ld-is-gold', 'ld-is-gold', '', d)}" = "ld-is-gold" ] ; then
                sed -i 's/$(CROSS_COMPILE)ld$/$(CROSS_COMPILE)ld.bfd/g' ${S}/config.mk
        fi

        unset LDFLAGS
        unset CFLAGS
        unset CPPFLAGS
        if [ -n "${EXTERNALSRC}" ] ; then
                export KBUILD_OUTPUT=${EXTERNALSRC}/${MACHINE}/
        fi
        export ${UBOOT_ARCH_ARGS} DEVICE_TREE=${UBOOT_DEVICE_TREE}
        oe_runmake ${UBOOT_MACHINE}
        if [ -n "${EXTERNALSRC}" ] ; then
                ${S}/scripts/kconfig/merge_config.sh -O ${KBUILD_OUTPUT} -m ${KBUILD_OUTPUT}.config ${@" ".join(find_cfgs(d))}
        else
                ${S}/scripts/kconfig/merge_config.sh -m .config ${@" ".join(find_cfgs(d))}
        fi
}

do_compile() {
        unset LDFLAGS
        unset CFLAGS
        unset CPPFLAGS
        if [ -n "${EXTERNALSRC}" ] ; then
                export KBUILD_OUTPUT=${EXTERNALSRC}/${MACHINE}/
        fi
        export ${UBOOT_ARCH_ARGS} DEVICE_TREE=${UBOOT_DEVICE_TREE}

        oe_runmake
}

do_install() {
        install -d ${D}/boot

        if [ -n "${EXTERNALSRC}" ] ; then
                export KBUILD_OUTPUT=${EXTERNALSRC}/${MACHINE}/
                install -m 0644 ${KBUILD_OUTPUT}${UBOOT_NAME}.${UBOOT_SUFFIX}   ${D}/boot/${UBOOT_IMAGE}
        else
                install -m 0644 ${S}/${UBOOT_NAME}.${UBOOT_SUFFIX}      ${D}/boot/${UBOOT_IMAGE}
        fi

        if ${@bb.utils.contains('INVITE_PLATFORM', 'early-camera', 'true', 'false', d)}; then
                install -m 0644 ${S}/drivers/camera/splash/${SPLASH_IMAGE}  ${D}/boot/${SPLASH_IMAGE}
        fi
}

FILES_${PN} = "/boot"

do_deploy() {
    install -d ${DEPLOYDIR}
    install -m 0644 ${D}/boot/${UBOOT_IMAGE} ${DEPLOYDIR}

        cd ${DEPLOYDIR}
    rm -f ${UBOOT_BINARY} ${UBOOT_SYMLINK}
    ln -sf ${UBOOT_IMAGE} ${UBOOT_BINARY}
    ln -sf ${UBOOT_IMAGE} ${UBOOT_SYMLINK}
        cd -

        if ${@bb.utils.contains('INVITE_PLATFORM', 'early-camera', 'true', 'false', d)}; then
        install -m 0644 ${D}/boot/${SPLASH_IMAGE} ${DEPLOYDIR}
    fi
}

do_clean_extworkdir() {
        if [ -n "${EXTERNALSRC}" ]; then
                rm -rf ${EXTERNALSRC}/arch/*/include/asm/telechips
                rm -rf ${EXTERNALSRC}/${MACHINE}
        fi
}

addtask deploy before do_build after do_install
addtask clean_extworkdir after do_buildclean before do_clean
```

Verwenden Sie ***devtool***, um den Quellcode **u-boot-tcc** in den Workspace herunterzuladen.

Mit der ***BitBake***-Option **-e** können Sie den Quellpfad wie folgt ermitteln:

```
$ devtool modify u-boot-tcc

$ bitbake u-boot-tcc -e | grep ^S=
S=”/home/TOPST/build/tcc8050-main/workspace/sources/u-boot-tcc”

$ cd /home/TOPST/build/tcc8050-main/workspace/sources/u-boot-tcc
$ ls

arch  cmd    config.mk  disk     drivers  env  fs        Kbuild   lib      MAINTAINERS  net       post            scripts  tools
```

<br/>

**U-boot-tcc** kann die Funktion **Cleanall** oder **build** über die Option ***BitBake* -c** verwenden. Die Funktion muss jedoch als Task im Rezept definiert sein.

```
$ bitbake u-boot-tcc -c cleanall

$ bitbake u-boot-tcc -c build
```

	```
	SRC_URI += “file://001-test-dts.patch”
	```

2.  U-Boot neu erstellen und ausführen.

	Wenn Sie die ***BitBake***-Option **-c** **build** verwenden, wird die U-Boot-Binärdatei nach der Kompilierung automatisch als Pfad für Binär-Downloads gespeichert.

	```
	$ bitbake u-boot-tcc -c cleansstate

	$ bitbake u-boot-tcc -c build

	$ ls ~/TOPST/build/tcc8050-main/workspace/sources/u-boot-tcc/ca72_bl3.rom
	ca72_bl3.rom
	```

```
# cat linux-telechips_%.bbappend

do_change_defconfig() {
	if ${@bb.utils.contains_any('SUBCORE_APPS', 'rvc svm', 'true', 'false', d)}; then

		echo "CONFIG_TCC_MULTI_MAILBOX=y"						>> ${WORKDIR}/defconfig
		echo "# CONFIG_VIDEO_TCC_VOUT is not set"				>> ${WORKDIR}/defconfig
		echo "CONFIG_SWITCH_REVERSE=y"							>> ${WORKDIR}/defconfig
		echo "CONFIG_FB_TCC_OVERLAY=y"							>> ${WORKDIR}/defconfig

		if ${@bb.utils.contains_any('SUBCORE_APPS', 'rvc', 'true', 'false', d)}; then
			echo "CONFIG_OVERLAY_PGL=y"							>> ${WORKDIR}/defconfig
		fi
	fi
		
	# TOPST : Raspberry Camera
	echo "CONFIG_VIDEO_IMX219=y"							>> ${WORKDIR}/defconfig
	echo "CONFIG_VIDEOBUF2_VMALLOC=y"						>> ${WORKDIR}/defconfig

	if ${@bb.utils.contains('SUBCORE_APPS', 't-sound', 'true', 'false', d)}; then

		echo "CONFIG_SND=y"										>>  ${WORKDIR}/defconfig

	fi

# network configuration for ssh server
	if ${@bb.utils.contains('INVITE_PLATFORM', 'network', 'true', 'false', d)}; then
		echo "CONFIG_INET=y"									>> ${WORKDIR}/defconfig
		echo "CONFIG_IPV6=y"									>> ${WORKDIR}/defconfig
		echo "CONFIG_NETDEVICES=y"								>> ${WORKDIR}/defconfig
```

### 4.3.6 I2C Master

I2C ist ein Master-Slave-Protokoll, bei dem die Kommunikation zwischen einem Host-Adapter (oder Host-Controller) und Client-Geräten (oder Slaves) stattfindet, und es verfügt außerdem über zwei Leitungen für den Takt und die bidirektionale Datenübertragung. Die entsprechenden Leitungen heißen Serial Clock (SCL) und Serial Data (SDA). Die I2C-Master-Geräteknoten (wie /dev/i2c-x) werden für den Zugriff aus dem User Space erstellt.

### 4.3.8.2.2 GPIO-sysfs-Schnittstelle

Der Kernel stellt über **sysfs** eine Schnittstelle bereit, um GPIO im User Space zu verwenden. Die Dokumentation zur Verwendung der **sysfs-Schnittstelle** von GPIO befindet sich in der Datei “Documentation/gpio/sysfs.txt” im Kernel.

Um **sysfs** einzubinden, fügen Sie Folgendes in **menuconfig** hinzu.

```
$ bitbake virtual/kernel -c menuconfig

Device Drivers ---> 
GPIO Support ---> 
[*] /sys/class/gpio/… (sysfs interface)
```

- GPIO-Nummer

Für den von Ihnen verwendeten Chip ist eine pinctrl-Datei verfügbar. Wenn Ihre Plattform TCC8050 ist, sollten Sie “tcc8050-pinctrl.dtsi” verwenden. Diese Datei definiert GPIO für die Verwendung von **gpio a/b/c/d/e** usw. Der Wert **reg** bestimmt den Wert in **sysfs**.

Der Wert von **gpio a** ist beispielsweise 32 und wird im Wert **reg** definiert. **gpio a** ist der erste GPIO, daher kann er in **sysfs** von 0 bis 31 eingegeben werden.

Wenn Sie **gpio b** übergeben, beginnt **gpio b 0** bei 32, da **gpio a** 0 bis 31 verwendet.

Wenn Sie beispielsweise **gpio b 3** verwenden, sollten Sie in **sysfs** 32 + 3 eingeben.

Die jedem GPIO zugewiesene GPIO-Nummer ist unten rot dargestellt. Der GPIO erhöht den GPIO-Wert von 0 auf 31 und gibt ihn in **sysfs** ein.

```
gpa: gpa { 
gpio-controller; 
#gpio-cells = <2>; 
reg = <0x0 0 32>; 
}; 

gpb: gpb { 
gpio-controller; 
#gpio-cells = <2>; 
reg = <0x40 0 32>; 
}; 

gpc: gpc { 
gpio-controller; 
#gpio-cells = <2>; 
reg = <0x80 0 32>; 
};

```

### 4.3.8.2.3 GPIO-dev-Schnittstelle

- Vorbereitung der GPIO-dev-Schnittstelle (**libgpiod**)

Die GPIO-sysfs-Schnittstelle ist veraltet und eine neue GPIO-dev-Schnittstelle wurde eingepflegt. Die GPIO-dev-Schnittstelle erfordert libgpiody, eine C-Bibliothek mit Werkzeugen für GPIO-chardev, und die GPIO-dev-Schnittstelle verwendet die Geräteknoten mit den Namen gpiochip0, gpiochip1, ghiochip2 usw. im Verzeichnis /dev.

Das folgende Beispiel ist einfach und deckt nur die grundlegenden Funktionen der GPIO-Steuerung ab.

- Beispiel für einen Raspberry-GPIO-Pin (**RPI GPIO_C22 PIN7**)

```c
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <fcntl.h>
#include <string.h>

// RPI GPIO_C22 PIN7

int main(int argc, char *argv[])
{
    int fd;
    char buf[64];
    int pin = 22; // choose a GPIO pin
    int value = 1; // choose a value (0 or 1)

    // open GPIO chardev number 0. GPIO group C. 
    snprintf(buf, sizeof(buf), "/dev/gpiochip2");
    fd = open(buf, O_RDWR);
    if (fd < 0) {
        perror("Failed to open gpiochip0");
        exit(1);
    }

// get the desired line of GPIO chip. GPIO_C22. 
    snprintf(buf, sizeof(buf), "gpio%d/direction", pin);
    if (write(fd, "out", 4) < 0) {
        perror("Failed to set direction of gpio pin");
        exit(1);
    }

    // set the value of the GPIO pin. GPIO_C22.
    snprintf(buf, sizeof(buf), "gpio%d/value", pin);
    if (write(fd, value ? "1" : "0", 2) < 0) {
        perror("Failed to set value of gpio pin");
        exit(1);
    }

    // close the GPIO chip device
    close(fd);

    return 0;
}

```

Acht Port-Kanäle (Port 0 bis Port 7)

Um die Aktivierung des UART-Treibers im Kernel festzulegen, führen Sie das folgende Kernel-Konfigurationswerkzeug aus:

```
$ bitbake virtual/kernel -c menuconfig

Device Drivers ---> 
GPIO Support ---> 
[*] Pulse-Width Modulation (PWM) Support
    <*> Telechips PWM support
```

<br/>

Nach dem Booten des Systems können Sie die Geräteknoten mit dem folgenden Befehl finden.

```
root@jammy:~# ls /sys/class/pwm/pwmchip0/
device  export  npwm  power  subsystem  uevent  unexport  waiting_for_supplier
```
